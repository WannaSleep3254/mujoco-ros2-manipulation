cmake_minimum_required(VERSION 3.16)
project(clock_action_startup_race LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
add_executable(clock_action_race clock_action_race.cpp)
target_compile_features(clock_action_race PRIVATE cxx_std_17)
target_compile_options(clock_action_race PRIVATE -Wall -Wextra -Werror -g)
target_link_options(clock_action_race PRIVATE -Wl,--export-dynamic)
target_link_libraries(clock_action_race dl pthread)
ament_target_dependencies(clock_action_race rclcpp rclcpp_action control_msgs)
